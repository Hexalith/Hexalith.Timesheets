import gzip
import hashlib
import json
from pathlib import Path
import re
import sys

if sys.flags.optimize:
    raise SystemExit('Run without Python optimization; this consistency check uses assertions')
bundle = Path(sys.argv[1]).resolve()
root = bundle.parents[3]

def read(name):
    path = bundle / name
    if not path.exists():
        path = bundle / (name + '.gz')
    raw = path.read_bytes()
    return gzip.decompress(raw) if path.suffix == '.gz' else raw

def data(name):
    return json.loads(read(name))

manifest = data('manifest.json')['files']
assert len({row['file'] for row in manifest}) == len(manifest)
actual = {str(path.relative_to(bundle)) for path in bundle.rglob('*') if path.is_file()}
assert actual == {row['file'] for row in manifest} | {'manifest.json'}
for row in manifest:
    encoded = (bundle / row['file']).read_bytes()
    assert hashlib.sha256(encoded).hexdigest() == row['sha256'], row['file']
    raw = gzip.decompress(encoded) if row['encoding'] == 'gzip' else encoded
    assert hashlib.sha256(raw).hexdigest() == row['original_sha256'], row['file']
    assert len(raw) == row['original_bytes'], row['file']

head = '670546c496192dfdd79be695de24200fe445488c'
checkout = data('checkout-identity.json')
assert checkout['head'] == head and checkout['branch'] == 'main'
assert len(checkout['root_gitlinks']) == 13
assert not checkout['root_alternates_file_exists']
for row in checkout['root_gitlinks']:
    assert row['actual_revision'] == row['commit']
    assert not row['status'] and not row['alternates_file_exists'] and not row['initialized_nested_paths']
for name in ('restore', 'build'):
    receipt = data(name + '.json')
    assert receipt['exit_code'] == 0 and receipt['source_revision'] == head
assert '0 Warning(s)' in read('build.log').decode() and '0 Error(s)' in read('build.log').decode()
assert data('native-run.json')['exit_code'] == 1
admission = data('runtime/run-y7do08u7/admission.json')
assert admission['reason'] == 'Required port 3501 is occupied; no AppHost started'
assert all(admission[key] is False for key in ('admitted', 'resources_started', 'infrastructure_started', 'apphost_start_attempted'))
cleanup = data('runtime/run-y7do08u7/cleanup.json')
assert cleanup['complete'] and cleanup['journey_descendants_gone'] and not cleanup['failure_reasons']
assert cleanup['containers_removed'] == 0 and not cleanup['components_created']
runtime_files = [row['file'] for row in manifest if row['file'].startswith('runtime/')]
assert len(runtime_files) == 7
for name in ('04-dapr-list.log', '05-aspire-ps.log'):
    assert 'Exit: 0\nStdout:\n[]\n' in read('runtime/run-y7do08u7/' + name).decode()
assets = data('restored-identities.json')
assert assets['sdk'] == '10.0.402' and len(assets['projects']) == 30
assert len(assets['checkout_retained_historical_assets']) == 17
assert not assets['release_output_directories']
for row in assets['projects']:
    assert Path(row['project_path']).is_relative_to(checkout['checkout'])
    assert all(Path(reference).is_relative_to(checkout['checkout']) for references in row['project_references'].values() for reference in references)
packages = {row['package'] for row in assets['packages']}
assert {'Aspire.Dashboard.Sdk.linux-x64/13.6.4', 'Aspire.Hosting.Orchestration.linux-x64/13.6.4'} <= packages
preserved = data('workspace-preservation.json')
assert preserved['matches'] and preserved['after'] == data('workspace-before.json')
processes = data('visible-processes-later.json')
assert processes['initial_observed_count'] == 547
assert processes['initial_identities_absent_count'] == len(processes['initial_identities_absent']) == 22
report = root / '_bmad-output/implementation-artifacts/tests/remaining-local-deferred-followups.md'
added = report.read_text().split('## Aspire 13.6.4 local qualification attempt', 1)[1]
for target in re.findall(r'\]\(([^)]+)\)', added):
    assert (report.parent / target.split('#')[0]).exists(), target
print(json.dumps({'verified_manifest_files': len(manifest), 'runtime_admitted': False,
                  'runtime_started': False, 'fresh_assets': 30, 'links_exist': True,
                  'scope': 'Consistency of retained records and local links; no independent runtime observation or tamper-proof attestation.'}))
