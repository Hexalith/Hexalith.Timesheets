"""Later transcription of inline collector logic; not the original executed file.

Read-only collection, with output only to a new caller-selected directory. This
can explain/reproduce collection logic on a later state, not recover historical
timestamps, process visibility, Git statuses, or raw /proc tables never retained.
No fixture, process lifecycle, build or restore is invoked.
"""

import argparse
import base64
import hashlib
import json
import os
from pathlib import Path
import subprocess
import time
import xml.etree.ElementTree as ET

parser = argparse.ArgumentParser()
parser.add_argument('--checkout', type=Path, required=True)
parser.add_argument('--output', type=Path, required=True)
args = parser.parse_args()
root = args.checkout.resolve()
args.output.mkdir(parents=True, exist_ok=False)

def git(path, *arguments):
    return subprocess.check_output(['git', '-C', str(path), *arguments], text=True).strip()

def write(name, value):
    (args.output / name).write_text(json.dumps(value, indent=2) + '\n')

links = [{'path': line.split('\t')[1], 'commit': line.split()[2]}
         for line in git(root, 'ls-tree', '-r', 'HEAD').splitlines() if line.startswith('160000')]
repositories = {'.': root, **{row['path']: root / row['path'] for row in links}}
identities = {}
for name, path in repositories.items():
    patch = subprocess.check_output(['git', '-C', str(path), 'diff', 'HEAD', '--no-ext-diff', '--binary'])
    untracked = git(path, 'ls-files', '--others', '--exclude-standard', '-z').split('\0')
    inputs = {}
    for value in untracked:
        if not value:
            continue
        relative = Path(value)
        if relative.parts[0] in ('src', 'tests', 'test', 'scripts') or len(relative.parts) == 1 and relative.suffix in ('.props', '.targets', '.csproj', '.slnx', '.json'):
            inputs[value] = hashlib.sha256((path / relative).read_bytes()).hexdigest()
    identities[name] = {'revision': git(path, 'rev-parse', 'HEAD'),
                        'tracked_diff_sha256': hashlib.sha256(patch).hexdigest(),
                        'status': git(path, 'status', '--porcelain=v1'), 'untracked_build_inputs': inputs}
static_files = {}
for project in ('src/Hexalith.Timesheets.AppHost', 'src/Hexalith.Timesheets', 'tests/Hexalith.Timesheets.IntegrationTests', 'references/Hexalith.EventStore/src/Hexalith.EventStore'):
    for path in (root / project / 'bin/Debug/net10.0').rglob('*'):
        if path.is_file():
            static_files[str(path.relative_to(root))] = hashlib.sha256(path.read_bytes()).hexdigest()
write('built-source-new-observation.json', {'observed_unix_ns': time.time_ns(), 'repositories': identities,
                                         'static_built_files_sha256': dict(sorted(static_files.items()))})
checkout_rows = []
for row in links:
    path = root / row['path']
    nested = [line.split('\t')[1] for line in git(path, 'ls-tree', '-r', 'HEAD').splitlines() if line.startswith('160000')]
    checkout_rows.append({**row, 'actual_revision': git(path, 'rev-parse', 'HEAD'),
                          'status': git(path, 'status', '--porcelain=v1'),
                          'git_directory': git(path, 'rev-parse', '--absolute-git-dir'),
                          'alternates_file_exists': (path / '.git/objects/info/alternates').exists(),
                          'nested_gitlink_count': len(nested),
                          'initialized_nested_paths': [name for name in nested if (path / name / '.git').exists()]})
write('checkout-new-observation.json', {'observed_unix_ns': time.time_ns(), 'checkout': str(root),
                                      'head': git(root, 'rev-parse', 'HEAD'), 'root_gitlinks': checkout_rows})
assets_path = root / 'src/Hexalith.Timesheets.AppHost/obj/project.assets.json'
assets = json.loads(assets_path.read_text())
packages = []
for key, library in assets['libraries'].items():
    if not key.startswith(('Aspire.', 'CommunityToolkit.Aspire.')):
        continue
    package, version = key.split('/')
    cache = next(Path(folder) / library['path'] for folder in assets['packageFolders'] if (Path(folder) / library['path']).exists())
    nuspec = next(cache.glob('*.nuspec'))
    repository = next((e.attrib for e in ET.parse(nuspec).iter() if e.tag.split('}')[-1] == 'repository'), {})
    files = [nuspec] + [cache / name for name in ('tools/Aspire.Dashboard', 'tools/dcp', 'tools/dcpctrl') if (cache / name).is_file()]
    row = {'package': key, 'repository': repository, 'assets_content_hash_sha512': library.get('sha512'),
           'files_sha256': {str(path.relative_to(cache)): hashlib.sha256(path.read_bytes()).hexdigest() for path in files}}
    archive = cache / (package.lower() + '.' + version + '.nupkg')
    if archive.exists():
        row['local_archive_sha512'] = base64.b64encode(hashlib.sha512(archive.read_bytes()).digest()).decode()
    sidecar = Path(str(archive) + '.sha512')
    if sidecar.exists():
        row['local_archive_sidecar_sha512'] = sidecar.read_text().strip()
    if (cache / '.nupkg.metadata').exists():
        row['metadata_content_hash_sha512'] = json.loads((cache / '.nupkg.metadata').read_text()).get('contentHash')
    packages.append(row)
fresh, historical = [], []
for path in root.rglob('project.assets.json'):
    data = json.loads(path.read_text())
    row = {'assets': str(path.relative_to(root)), 'sha256': hashlib.sha256(path.read_bytes()).hexdigest(),
           'project_path': data['project']['restore']['projectPath'],
           'project_references': {tfm: list(value.get('projectReferences', {})) for tfm, value in data['project']['restore']['frameworks'].items()},
           'project_libraries': sorted(key for key, value in data['libraries'].items() if value['type'] == 'project')}
    (fresh if Path(row['project_path']).is_relative_to(root) else historical).append(row)
write('restored-new-observation.json', {'apphost_assets_sha256': hashlib.sha256(assets_path.read_bytes()).hexdigest(),
                                     'packages': packages, 'projects': sorted(fresh, key=lambda row: row['assets']),
                                     'checkout_retained_historical_assets': sorted(historical, key=lambda row: row['assets'])})
listeners = []
errors = {'socket_tables': 0, 'fd_directories': 0, 'fd_links': 0, 'process_identity': 0}
for family, filename in (('ipv4', '/proc/net/tcp'), ('ipv6', '/proc/net/tcp6')):
    try:
        for line in Path(filename).read_text().splitlines()[1:]:
            fields = line.split()
            address, port = fields[1].rsplit(':', 1)
            if int(port, 16) == 3501 and fields[3] == '0A':
                listeners.append({'family': family, 'local_address_hex': address, 'port': 3501,
                                  'inode': fields[9], 'visible_holders': []})
    except OSError:
        errors['socket_tables'] += 1
inodes = {row['inode'] for row in listeners}
for path in Path('/proc').iterdir():
    if not path.name.isdigit():
        continue
    try:
        descriptors = list((path / 'fd').iterdir())
    except OSError:
        errors['fd_directories'] += 1
        continue
    for descriptor in descriptors:
        try:
            link = os.readlink(descriptor)
        except OSError:
            errors['fd_links'] += 1
            continue
        if not (link.startswith('socket:[') and link[8:-1] in inodes):
            continue
        try:
            fields = (path / 'stat').read_text().rsplit(')', 1)[1].split()
            identity = {'pid': int(path.name), 'start': fields[19], 'name': (path / 'comm').read_text().strip(), 'executable': os.readlink(path / 'exe')}
            assert (path / 'stat').read_text().rsplit(')', 1)[1].split()[19] == identity['start']
        except (OSError, IndexError, AssertionError):
            errors['process_identity'] += 1
            continue
        for row in listeners:
            if row['inode'] == link[8:-1] and identity not in row['visible_holders']:
                row['visible_holders'].append(identity)
write('listener-new-observation.json', {'observed_unix_ns': time.time_ns(), 'port': 3501,
                                      'listeners': listeners, 'visibility_errors': errors})
states = {'01': 'ESTABLISHED', '02': 'SYN_SENT', '03': 'SYN_RECV', '04': 'FIN_WAIT1', '05': 'FIN_WAIT2', '06': 'TIME_WAIT', '07': 'CLOSE', '08': 'CLOSE_WAIT', '09': 'LAST_ACK', '0A': 'LISTEN', '0B': 'CLOSING'}
rows = []
for family, filename in (('ipv4', '/proc/net/tcp'), ('ipv6', '/proc/net/tcp6'), ('udp4', '/proc/net/udp'), ('udp6', '/proc/net/udp6')):
    for line in Path(filename).read_text().splitlines()[1:]:
        fields = line.split()
        if int(fields[1].rsplit(':', 1)[1], 16) == 3501:
            rows.append({'family': family, 'local_address_hex': fields[1].rsplit(':', 1)[0], 'local_port': 3501,
                         'state_code': fields[3], 'tcp_state_name': states.get(fields[3]) if family.startswith('ipv') else None, 'inode': fields[9]})
write('all-port-states-new-observation.json', {'observed_unix_ns': time.time_ns(), 'rows': rows})
