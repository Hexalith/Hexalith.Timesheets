empty_sha = hashlib.sha256(b'').hexdigest()

def read(name):
    path = bundle / name
    if not path.exists():
        path = bundle / (name + '.gz')
    raw = path.read_bytes()
    return gzip.decompress(raw) if path.suffix == '.gz' else raw

def data(name):
    return json.loads(read(name))

def safe_path(value, base, relative=False):
    path = Path(value)
    assert '..' not in path.parts, value
    assert not relative or not path.is_absolute(), value
    resolved = (base / path).resolve() if relative else path.resolve()
    assert resolved.is_relative_to(base.resolve()), value
    return resolved

def git(*arguments):
    return subprocess.check_output(['git', '-C', str(root), *arguments], text=True).strip()

    raw = gzip.decompress(encoded) if row['encoding'] == 'gzip' else encoded
